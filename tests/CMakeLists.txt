function(zcgvs_add_kernel_test name source)
    add_executable(zcgvs_test_${name} ${source})
    target_link_libraries(zcgvs_test_${name} PRIVATE zcgvs_core)
    target_include_directories(zcgvs_test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpp)
    add_test(NAME ${name} COMMAND zcgvs_test_${name})
endfunction()

zcgvs_add_kernel_test(similarity cpp/test_similarity.cu)
zcgvs_add_kernel_test(topk cpp/test_topk.cu)
zcgvs_add_kernel_test(metrics_filter cpp/test_metrics_filter.cu)

zcgvs_add_kernel_test(topk_large cpp/test_topk_large.cu)

zcgvs_add_kernel_test(pipeline cpp/test_pipeline.cu)
zcgvs_add_kernel_test(lexical cpp/test_lexical.cu)
